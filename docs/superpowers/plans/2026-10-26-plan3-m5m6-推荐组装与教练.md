```ts
export function normalizeText(s: string): string {
  return s.toLowerCase().replace(/\s+/g, ' ').trim();
}

export function isHit(haystack: string, term: string): boolean {
  const h = normalizeText(haystack);
  const t = normalizeText(term);
  if (!t) return false;
  return h.includes(t) || h.includes(t.replace(/\s+/g, ''));
}

export function computeHits(text: string, terms: string[]): { hits: string[]; missing: string[]; coveragePct: number } {
  const hits: string[] = [];
  const missing: string[] = [];
  for (const t of terms) (isHit(text, t) ? hits : missing).push(t);
  const total = terms.length;
  return { hits, missing, coveragePct: total === 0 ? 0 : Math.round((hits.length / total) * 100) };
}
```

```ts
function assembleResume(db: DatabaseSync, jd: JdRow) {
  const chosen = JSON.parse(jd.chosen_items) as number[];
  const overrides = JSON.parse(jd.overrides) as Record<string, { bullets: string[] }>;
  const rows = db.prepare('SELECT * FROM profile_items ORDER BY section, sort_order, id').all() as unknown as ItemRow[];
  const items = rows.filter((r) => chosen.includes(r.id)).map((r) => {
    const base = JSON.parse(r.bullets) as string[];
    const ov = overrides[String(r.id)];
    return { item_id: r.id, section: r.section, heading: r.heading, period: r.period, bullets: ov ? ov.bullets : base };
  });
  return { items };
}

  r.get('/:id/resume', (req, res) => {
    const jd = db.prepare('SELECT * FROM jds WHERE id=?').get(Number(req.params.id)) as JdRow | undefined;
    if (!jd) return fail(res, 404, 'JD 不存在');
    ok(res, assembleResume(db, jd));
  });

  r.post('/:id/match', (req, res) => {
    const id = Number(req.params.id);
    const jd = db.prepare('SELECT * FROM jds WHERE id=?').get(id) as JdRow | undefined;
    if (!jd) return fail(res, 404, 'JD 不存在');
    const resume = assembleResume(db, jd);
    const text = resume.items.map((i) => i.heading + ' ' + i.bullets.join(' ')).join(' ');
    const kws = db.prepare('SELECT k.id, k.term FROM keywords k JOIN jd_keywords jk ON jk.keyword_id=k.id WHERE jk.jd_id=?').all(id) as unknown as { id: number; term: string }[];
    const result = computeHits(text, kws.map((k) => k.term));
    const ins = db.prepare('INSERT INTO keyword_hits (jd_id, keyword_id, status) VALUES (?,?,?) ON CONFLICT(jd_id, keyword_id) DO UPDATE SET status=excluded.status');
    for (const k of kws) ins.run(id, k.id, isHit(text, k.term) ? 'hit' : 'missing');
    ok(res, { ...result, total: kws.length });
  });

  r.get('/:id/export', (req, res) => {
    const jd = db.prepare('SELECT * FROM jds WHERE id=?').get(Number(req.params.id)) as JdRow | undefined;
    if (!jd) return fail(res, 404, 'JD 不存在');
    ok(res, { exportedAt: nowIso(), jd: { company: jd.company, title: jd.title }, resume: assembleResume(db, jd) });
  });
```

```ts
import { Router } from 'express';
import { DatabaseSync } from 'node:sqlite';
import { ok, fail } from '../http';
import { nowIso, takeVersion, ItemRow } from '../db';
import { getAIConfig, chatJson } from '../ai/client';

interface RunRow { id: number; jd_id: number; item_id: number; original_bullets: string; suggestions: string; backfilled: number; created_at: string; updated_at: string }
interface Suggestion { bullet_index: number; new_text: string; star_label: string; reason: string; status: string }
interface JdRowLite { id: number; company: string; title: string; summary: string; raw_text: string; overrides: string }

function syncOverrides(db: DatabaseSync, run: RunRow): void {
  const jd = db.prepare('SELECT * FROM jds WHERE id=?').get(run.jd_id) as JdRowLite;
  const overrides = JSON.parse(jd.overrides) as Record<string, { bullets: string[] }>;
  const base = JSON.parse(run.original_bullets) as string[];
  const sug = JSON.parse(run.suggestions) as Suggestion[];
  const bullets = [...base];
  for (const s of sug) {
    if (s.status === 'accepted' && s.bullet_index >= 0 && s.bullet_index < bullets.length) bullets[s.bullet_index] = s.new_text;
  }
  overrides[String(run.item_id)] = { bullets };
  db.prepare('UPDATE jds SET overrides=?, updated_at=? WHERE id=?').run(JSON.stringify(overrides), nowIso(), run.jd_id);
}

export function coachRouter(db: DatabaseSync): Router {
  const r = Router();

  r.post('/jds/:jdId/coach/:itemId', async (req, res) => {
    const jdId = Number(req.params.jdId);
    const itemId = Number(req.params.itemId);
    const jd = db.prepare('SELECT * FROM jds WHERE id=?').get(jdId) as JdRowLite | undefined;
    const item = db.prepare('SELECT * FROM profile_items WHERE id=?').get(itemId) as ItemRow | undefined;
    if (!jd || !item) return fail(res, 404, 'JD 或履历条目不存在');
    const bullets = JSON.parse(item.bullets) as string[];
    if (!bullets.length) return fail(res, 400, '该条目还没有亮点，先补充内容再优化');
    try {
      const cfg = getAIConfig(db);
      const out = await chatJson(cfg, [
        { role: 'system', content: '你是 STAR 简历教练。针对 JD 要求逐条改写候选人的亮点 bullet：动词开头、覆盖情境/任务/行动/结果、尽量量化、每条不超过 80 字、不编造事实（无数据时用定性描述）。只输出 JSON：{"suggestions":[{"bullet_index":数字,"new_text":"改写后","star_label":"补强了 S/T/A/R 中哪部分","reason":"中文理由"}]}' },
        { role: 'user', content: `【JD】${jd.company} ${jd.title}\n要点：${jd.summary}\n【条目】${item.heading}\n${bullets.map((b, i) => `[${i}] ${b}`).join('\n')}` },
      ]);
      const raw = Array.isArray(out?.suggestions) ? out.suggestions : [];
      const suggestions: Suggestion[] = raw
        .filter((s: { bullet_index?: unknown }) => Number.isInteger(Number(s.bullet_index)) && Number(s.bullet_index) >= 0 && Number(s.bullet_index) < bullets.length)
        .map((s: { bullet_index: number; new_text: string; star_label?: string; reason?: string }) => ({
          bullet_index: Number(s.bullet_index), new_text: String(s.new_text ?? '').slice(0, 300),
          star_label: String(s.star_label ?? ''), reason: String(s.reason ?? ''), status: 'pending',
        }));
      if (!suggestions.length) return fail(res, 502, '模型未返回有效改写建议，请重试');
      const t = nowIso();
      db.prepare('DELETE FROM coach_runs WHERE jd_id=? AND item_id=?').run(jdId, itemId);
      const info = db.prepare('INSERT INTO coach_runs (jd_id, item_id, original_bullets, suggestions, backfilled, created_at, updated_at) VALUES (?,?,?,?,0,?,?)')
        .run(jdId, itemId, JSON.stringify(bullets), JSON.stringify(suggestions), t, t);
      const run = db.prepare('SELECT * FROM coach_runs WHERE id=?').get(Number(info.lastInsertRowid)) as unknown as RunRow;
      ok(res, { ...run, original_bullets: JSON.parse(run.original_bullets), suggestions: JSON.parse(run.suggestions) });
    } catch (e) { fail(res, 502, (e as Error).message); }
  });

  r.get('/runs/:id', (req, res) => {
    const run = db.prepare('SELECT * FROM coach_runs WHERE id=?').get(Number(req.params.id)) as RunRow | undefined;
    if (!run) return fail(res, 404, '教练记录不存在');
    ok(res, { ...run, original_bullets: JSON.parse(run.original_bullets), suggestions: JSON.parse(run.suggestions) });
  });

  r.put('/:id/decide', (req, res) => {
    const run = db.prepare('SELECT * FROM coach_runs WHERE id=?').get(Number(req.params.id)) as RunRow | undefined;
    if (!run) return fail(res, 404, '教练记录不存在');
    const idx = Number(req.body?.bullet_index);
    const status = String(req.body?.status);
    if (!['accepted', 'rejected', 'pending'].includes(status)) return fail(res, 400, 'status 必须 accepted/rejected/pending');
    const sug = JSON.parse(run.suggestions) as Suggestion[];
    const s = sug.find((x) => x.bullet_index === idx);
    if (!s) return fail(res, 404, '建议不存在');
    s.status = status;
    db.prepare('UPDATE coach_runs SET suggestions=?, updated_at=? WHERE id=?').run(JSON.stringify(sug), nowIso(), run.id);
    syncOverrides(db, { ...run, suggestions: JSON.stringify(sug) });
    ok(res, { id: run.id, suggestions: sug });
  });

  r.post('/:id/backfill', (req, res) => {
    const run = db.prepare('SELECT * FROM coach_runs WHERE id=?').get(Number(req.params.id)) as RunRow | undefined;
    if (!run) return fail(res, 404, '教练记录不存在');
    const accepted = (JSON.parse(run.suggestions) as Suggestion[]).filter((s) => s.status === 'accepted');
    if (!accepted.length) return fail(res, 400, '还没有采纳任何建议');
    const base = JSON.parse(run.original_bullets) as string[];
    for (const s of accepted) base[s.bullet_index] = s.new_text;
    db.prepare('UPDATE profile_items SET bullets=?, updated_at=? WHERE id=?').run(JSON.stringify(base), nowIso(), run.item_id);
    takeVersion(db, run.item_id, 'from_backfill');
    db.prepare('UPDATE coach_runs SET backfilled=1, updated_at=? WHERE id=?').run(nowIso(), run.id);
    ok(res, { backfilled: true, bullets: base });
  });

  r.post('/:id/material', (req, res) => {
    const run = db.prepare('SELECT * FROM coach_runs WHERE id=?').get(Number(req.params.id)) as RunRow | undefined;
    if (!run) return fail(res, 404, '教练记录不存在');
    const accepted = (JSON.parse(run.suggestions) as Suggestion[]).filter((s) => s.status === 'accepted');
    if (!accepted.length) return fail(res, 400, '还没有采纳任何建议');
    const jd = db.prepare('SELECT company, title FROM jds WHERE id=?').get(run.jd_id) as { company: string; title: string };
    const t = nowIso();
    const info = db.prepare('INSERT INTO materials (title, content, tags, source, created_at, updated_at) VALUES (?,?,?,?,?,?)')
      .run(`${jd.company}${jd.title} 改写亮点`, accepted.map((s) => s.new_text).join('\n'), JSON.stringify([jd.title]), 'from_coach', t, t);
    ok(res, { materialId: Number(info.lastInsertRowid), count: accepted.length });
  });

  return r;
}
```

```tsx
import { useEffect, useState } from 'react';
import { api } from '../api';
import { ResumeItemView, MatchResult, SECTION_LABELS, Section, Keyword } from '../types';

export default function ResumePreview({ jdId }: { jdId: number }) {
  const [items, setItems] = useState<ResumeItemView[]>([]);
  const [match, setMatch] = useState<MatchResult | null>(null);
  const [kws, setKws] = useState<Keyword[]>([]);
  const [err, setErr] = useState('');
  const [busy, setBusy] = useState(false);

  async function load() {
    try {
      const r = await api.jdResume(jdId);
      setItems(r.items);
      setKws(await api.listKeywords());
    } catch (e) { setErr((e as Error).message); }
  }
  useEffect(() => { load(); }, [jdId]);

  async function runMatch() {
    setBusy(true); setErr('');
    try { setMatch(await api.matchJd(jdId)); } catch (e) { setErr((e as Error).message); } finally { setBusy(false); }
  }
  async function exportJson() {
    try {
      const data = await api.exportJd(jdId);
      const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
      const a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = `职位定制简历-${data.jd.company || '未命名'}.json`;
      a.click();
      URL.revokeObjectURL(a.href);
    } catch (e) { setErr((e as Error).message); }
  }

  const hitSet = new Set(match?.hits ?? []);
  function highlight(text: string): React.ReactNode {
    if (!match) return text;
    let out: React.ReactNode[] = [text];
    for (const term of match.hits) {
      const next: React.ReactNode[] = [];
      for (const node of out) {
        if (typeof node !== 'string') { next.push(node); continue; }
        const lower = node.toLowerCase();
        const t = term.toLowerCase();
        let idx = lower.indexOf(t);
        if (idx < 0) { next.push(node); continue; }
        let rest = node;
        while (idx >= 0) {
          next.push(rest.slice(0, idx));
          next.push(<mark key={`${term}-${idx}-${next.length}`}>{rest.slice(idx, idx + term.length)}</mark>);
          rest = rest.slice(idx + term.length);
          idx = rest.toLowerCase().indexOf(t);
        }
        next.push(rest);
      }
      out = next;
    }
    return out;
  }

  const groups = ['work', 'project', 'education', 'skill', 'award'] as Section[];

  return (
    <div>
      {err && <div className="toast toast-err">{err}<button onClick={() => setErr('')}>×</button></div>}
      <div className="row">
        <button className="primary" onClick={runMatch} disabled={busy}>{busy ? '计算中…' : '重新计算关键词命中'}</button>
        <button onClick={exportJson}>导出定制简历 JSON</button>
        <span className="wide" />
        {match && <span>覆盖率 {match.coveragePct}%（{match.hits.length}/{match.total}）</span>}
      </div>
      {match && match.missing.length > 0 && (
        <div className="card">
          <b>JD 要求但简历未体现的词条（{match.missing.length}）：</b>
          <div className="row" style={{ flexWrap: 'wrap' }}>{match.missing.map((m) => <span key={m} className="tag tag-miss">{m}</span>)}</div>
        </div>
      )}
      <div className="card resume-paper">
        {groups.map((g) => {
          const its = items.filter((i) => i.section === g);
          if (!its.length) return null;
          return (
            <section key={g}>
              <h4>{SECTION_LABELS[g]}</h4>
              {its.map((it) => (
                <div key={it.item_id} className="resume-item">
                  <div className="resume-heading">{highlight(it.heading)}{it.period ? <span className="period">{it.period}</span> : null}</div>
                  <ul>{it.bullets.map((b, i) => <li key={i}>{highlight(b)}</li>)}</ul>
                </div>
              ))}
            </section>
          );
        })}
        {items.length === 0 && <p style={{ color: '#888' }}>还没有选择经历——先去"推荐经历"勾选几条</p>}
      </div>
      {kws.length === 0 && <p style={{ color: '#888', fontSize: 13 }}>提示：词条为空，先在 JD 列表点"解析"生成词条，命中率才有意义</p>}
    </div>
  );
}
```

```tsx
function CoachPanel({ jd, onError, onRefresh }: { jd: JdRow; onError: (m: string) => void; onRefresh: () => void }) {
  const [runs, setRuns] = useState<Record<number, { id: number; original_bullets: string[]; suggestions: Suggestion[]; backfilled: number }>>({});
  const [busyId, setBusyId] = useState<number | null>(null);

  async function optimize(itemId: number) {
    setBusyId(itemId);
    try {
      const run = await api.coach(jd.id, itemId);
      setRuns({ ...runs, [itemId]: run });
    } catch (e) { onError((e as Error).message); } finally { setBusyId(null); }
  }
  async function decide(itemId: number, runId: number, bulletIndex: number, status: string) {
    try {
      const r = await api.decide(runId, bulletIndex, status);
      setRuns({ ...runs, [itemId]: { ...runs[itemId], suggestions: r.suggestions } });
    } catch (e) { onError((e as Error).message); }
  }
  async function doBackfill(itemId: number, runId: number) {
    try { await api.backfill(runId); onError('已回填到母版履历 ✅（原版本可在历史版本中找回）'); setRuns({ ...runs, [itemId]: { ...runs[itemId], backfilled: 1 } }); onRefresh(); }
    catch (e) { onError((e as Error).message); }
  }
  async function toMaterial(itemId: number, runId: number) {
    try { const r = await api.coachToMaterial(runId); onError(`已沉淀 ${r.count} 条到素材库 ✅`); }
    catch (e) { onError((e as Error).message); }
  }

  if (jd.chosen_items.length === 0) return <p style={{ color: '#888' }}>先在"① 推荐经历"页勾选入简历的经历条目</p>;

  return (
    <div>
      {jd.chosen_items.map((itemId) => {
        const run = runs[itemId];
        return (
          <div key={itemId} className="card">
            <div className="row">
              <b className="wide">条目 #{itemId}</b>
              <button className="primary" onClick={() => optimize(itemId)} disabled={busyId === itemId}>{busyId === itemId ? 'AI 改写中…' : run ? '重新生成建议' : 'AI 按 JD 优化'}</button>
            </div>
            {run && run.suggestions.map((s) => (
              <div key={s.bullet_index} className="sug">
                <div className="orig">原文：{run.original_bullets[s.bullet_index]}</div>
                <div className="new">建议：{s.new_text}</div>
                <div className="meta">{s.star_label} · {s.reason}</div>
                <div className="row">
                  <button className={s.status === 'accepted' ? 'primary' : ''} onClick={() => decide(itemId, run.id, s.bullet_index, s.status === 'accepted' ? 'pending' : 'accepted')}>{s.status === 'accepted' ? '✓ 已采纳' : '采纳'}</button>
                  <button className={s.status === 'rejected' ? 'danger' : ''} onClick={() => decide(itemId, run.id, s.bullet_index, s.status === 'rejected' ? 'pending' : 'rejected')}>{s.status === 'rejected' ? '✕ 已拒绝' : '拒绝'}</button>
                </div>
              </div>
            ))}
            {run && (
              <div className="row">
                <button onClick={() => doBackfill(itemId, run.id)} disabled={run.backfilled === 1}>{run.backfilled === 1 ? '已回填母版' : '回填到母版履历'}</button>
                <button onClick={() => toMaterial(itemId, run.id)}>采纳内容存素材库</button>
              </div>
            )}
          </div>
        );
      })}
    </div>
  );
}
```
