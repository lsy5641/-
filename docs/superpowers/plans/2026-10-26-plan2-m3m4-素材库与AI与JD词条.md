```ts
import { DatabaseSync } from 'node:sqlite';
import { getSetting } from '../db';

export interface AIConfig { baseUrl: string; apiKey: string; model: string; temperature: number }
export interface ChatMsg { role: 'system' | 'user' | 'assistant'; content: string }

export class AIError extends Error {}

export function getAIConfig(db: DatabaseSync): AIConfig {
  return {
    baseUrl: getSetting(db, 'ai.baseUrl', 'https://api.deepseek.com'),
    apiKey: getSetting(db, 'ai.apiKey', ''),
    model: getSetting(db, 'ai.model', 'deepseek-chat'),
    temperature: Number(getSetting(db, 'ai.temperature', '0.7')),
  };
}

export async function chat(cfg: AIConfig, messages: ChatMsg[]): Promise<string> {
  if (!cfg.apiKey) throw new AIError('尚未配置 API Key：点右上角齿轮⚙ 填入');
  let res: Response;
  try {
    res = await fetch(cfg.baseUrl.replace(/\/+$/, '') + '/chat/completions', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${cfg.apiKey}` },
      body: JSON.stringify({ model: cfg.model, temperature: cfg.temperature, messages, stream: false }),
      signal: AbortSignal.timeout(120000),
    });
  } catch {
    throw new AIError('无法连接 AI 服务，请检查联网状态与 Base URL');
  }
  if (res.status === 401 || res.status === 403) throw new AIError('API Key 无效或已过期（HTTP ' + res.status + '）');
  if (res.status === 429) throw new AIError('触发限流或余额不足，稍后再试（HTTP 429）');
  if (!res.ok) throw new AIError('AI 服务返回错误 HTTP ' + res.status);
  const j = await res.json() as { choices?: { message?: { content?: string } }[] };
  const content = j.choices?.[0]?.message?.content;
  if (typeof content !== 'string') throw new AIError('AI 返回结构异常');
  return content;
}

export function parseJsonLoose(text: string): unknown {
  let t = text.trim();
  const fence = t.match(/```(?:json)?\s*([\s\S]*?)\s*```/);
  if (fence) t = fence[1].trim();
  const a = t.indexOf('{'), b = t.lastIndexOf('}');
  const c = t.indexOf('['), d = t.lastIndexOf(']');
  if (a >= 0 && (c < 0 || a < c)) t = t.slice(a, b + 1); else if (c >= 0) t = t.slice(c, d + 1);
  return JSON.parse(t);
}

export async function chatJson(cfg: AIConfig, messages: ChatMsg[]): Promise<any> {
  const first = await chat(cfg, messages);
  try { return parseJsonLoose(first); } catch { /* 带错重问一次 */ }
  const retry = await chat(cfg, [...messages,
    { role: 'assistant', content: first.slice(0, 2000) },
    { role: 'user', content: '你上一条输出不是合法 JSON，解析失败。请只重新输出修正后的 JSON，不要任何解释文字。' }]);
  try { return parseJsonLoose(retry); } catch { throw new AIError('模型输出格式异常，请重试或更换模型'); }
}
```

```ts
import { ok, fail } from '../http';
import { getSetting, setSetting } from '../db';

export function settingsRouter(db: DatabaseSync): Router {

    const keyMask = cfg.apiKey ? cfg.apiKey.slice(0, 4) + '…' + cfg.apiKey.slice(-4) : '';

    if (ai.model !== undefined && ai.model !== '') setSetting(db, 'ai.model', String(ai.model).trim());

    if (ai.apiKey !== undefined && ai.apiKey !== '') setSetting(db, 'ai.apiKey', String(ai.apiKey).trim());
    ok(res, { saved: true });

  r.post('/test', async (_req, res) => {
    try {
      const cfg = getAIConfig(db);

      ok(res, { reply: out.slice(0, 50) });
```

```ts
import { Router } from 'express';
import { DatabaseSync } from 'node:sqlite';
import { ok, fail } from '../http';
import { nowIso } from '../db';

interface MatRow { id: number; title: string; content: string; tags: string; source: string; created_at: string; updated_at: string }
function parse(row: MatRow) { return { ...row, tags: JSON.parse(row.tags) }; }

export function materialsRouter(db: DatabaseSync): Router {
  const r = Router();

  r.get('/', (_req, res) => {
    const rows = db.prepare('SELECT * FROM materials ORDER BY id DESC').all() as unknown as MatRow[];
    ok(res, rows.map(parse));
  });

  r.post('/', (req, res) => {
    const { title = '', content = '', tags = [] } = req.body ?? {};
    if (!Array.isArray(tags) || tags.some((x: unknown) => typeof x !== 'string')) return fail(res, 400, 'tags 必须是字符串数组');
    const t = nowIso();
    const info = db.prepare('INSERT INTO materials (title, content, tags, source, created_at, updated_at) VALUES (?,?,?,?,?,?)')
      .run(String(title), String(content), JSON.stringify(tags), 'manual', t, t);
    ok(res, parse(db.prepare('SELECT * FROM materials WHERE id=?').get(Number(info.lastInsertRowid)) as unknown as MatRow));
  });

  r.put('/:id', (req, res) => {
    const id = Number(req.params.id);
    const cur = db.prepare('SELECT * FROM materials WHERE id=?').get(id) as MatRow | undefined;
    if (!cur) return fail(res, 404, '素材不存在');
    const body = req.body ?? {};
    if (body.tags !== undefined && !(Array.isArray(body.tags) && body.tags.every((x: unknown) => typeof x === 'string'))) return fail(res, 400, 'tags 必须是字符串数组');
    db.prepare('UPDATE materials SET title=?, content=?, tags=?, updated_at=? WHERE id=?')
      .run(body.title ?? cur.title, body.content ?? cur.content, JSON.stringify(body.tags ?? JSON.parse(cur.tags)), nowIso(), id);
    ok(res, parse(db.prepare('SELECT * FROM materials WHERE id=?').get(id) as unknown as MatRow));
  });

  r.delete('/:id', (req, res) => {
    const info = db.prepare('DELETE FROM materials WHERE id=?').run(Number(req.params.id));
    ok(res, { deleted: info.changes });
  });

  return r;
}
```

```ts
import { Router } from 'express';
import { DatabaseSync } from 'node:sqlite';
import { ok, fail } from '../http';
import { getAIConfig, chatJson } from '../ai/client';

export function aiRouter(db: DatabaseSync): Router {
  const r = Router();

  r.post('/bullets', async (req, res) => {
    const text = String(req.body?.text ?? '').trim();
    if (text.length < 10) return fail(res, 400, '请粘贴至少 10 个字的流水账');
    try {
      const cfg = getAIConfig(db);
      const out = await chatJson(cfg, [
        { role: 'system', content: '你是简历教练。把用户的工作流水账拆成 3~8 条 STAR 风格的简历亮点 bullet：动词开头、有方法、尽量量化结果。只输出 JSON：{"bullets":["..."]}' },
        { role: 'user', content: text },
      ]);
      const bullets = Array.isArray(out?.bullets) ? out.bullets.filter((b: unknown) => typeof b === 'string') : [];
      if (!bullets.length) return fail(res, 502, '模型未返回有效亮点，请重试');
      ok(res, { bullets: bullets.slice(0, 8) });
    } catch (e) { fail(res, 502, (e as Error).message); }
  });

  return r;
}
```

```ts
import { Router } from 'express';
import { DatabaseSync } from 'node:sqlite';
import { ok, fail } from '../http';
import { nowIso } from '../db';

interface JdRow { id: number; company: string; title: string; raw_text: string; summary: string; analyzed: number; recommended: string; chosen_items: string; overrides: string; status: string; created_at: string; updated_at: string }
function parse(row: JdRow) { return { ...row, summary: JSON.parse(row.summary), recommended: JSON.parse(row.recommended), chosen_items: JSON.parse(row.chosen_items), overrides: JSON.parse(row.overrides) }; }

export function jdsRouter(db: DatabaseSync): Router {
  const r = Router();

  r.get('/', (_req, res) => {
    const rows = db.prepare('SELECT * FROM jds ORDER BY id DESC').all() as unknown as JdRow[];
    ok(res, rows.map(parse));
  });

  r.get('/:id', (req, res) => {
    const row = db.prepare('SELECT * FROM jds WHERE id=?').get(Number(req.params.id)) as JdRow | undefined;
    if (!row) return fail(res, 404, 'JD 不存在');
    ok(res, parse(row));
  });

  r.post('/', (req, res) => {
    const { company = '', title = '', raw_text = '' } = req.body ?? {};
    if (String(raw_text).trim().length < 30) return fail(res, 400, 'JD 原文太短（至少 30 字），请粘贴完整职位描述');
    const t = nowIso();
    const info = db.prepare('INSERT INTO jds (company, title, raw_text, created_at, updated_at) VALUES (?,?,?,?,?)')
      .run(String(company).trim(), String(title).trim(), String(raw_text), t, t);
    ok(res, parse(db.prepare('SELECT * FROM jds WHERE id=?').get(Number(info.lastInsertRowid)) as unknown as JdRow));
  });

  r.delete('/:id', (req, res) => {
    const id = Number(req.params.id);
    db.prepare('DELETE FROM jd_keywords WHERE jd_id=?').run(id);
    db.prepare('DELETE FROM keyword_hits WHERE jd_id=?').run(id);

    ok(res, { deleted: info.changes });
  });
```

```ts
function strArr(v: unknown): string[] { return Array.isArray(v) ? v.filter((x) => typeof x === 'string') : []; }

export function upsertKeywords(db: DatabaseSync, jdId: number, kws: { term: string; category: string; frequency: number }[]): void {
  const t = nowIso();
  for (const k of kws) {
    const term = String(k.term ?? '').trim();
    if (!term) continue;
    const cat = ['hard_skill', 'soft_skill', 'business'].includes(k.category) ? k.category : 'hard_skill';
    const freq = Number.isFinite(k.frequency) && k.frequency > 0 ? Math.round(k.frequency) : 1;
    db.prepare('INSERT INTO keywords (term, category, created_at, updated_at) VALUES (?,?,?,?) ON CONFLICT(term) DO UPDATE SET category=excluded.category, updated_at=excluded.updated_at').run(term, cat, t, t);
```

```ts
import { Router } from 'express';
import { DatabaseSync } from 'node:sqlite';
import { ok, fail } from '../http';
import { nowIso } from '../db';

const CATS = ['hard_skill', 'soft_skill', 'business'];

export function keywordsRouter(db: DatabaseSync): Router {
  const r = Router();

  r.get('/', (_req, res) => {
    const rows = db.prepare(`SELECT k.*, COALESCE((SELECT SUM(frequency) FROM jd_keywords jk WHERE jk.keyword_id=k.id),0) AS totalFrequency,
      (SELECT COUNT(*) FROM jd_keywords jk2 WHERE jk2.keyword_id=k.id) AS jdCount FROM keywords k ORDER BY totalFrequency DESC, k.id ASC`).all() as unknown as Record<string, unknown>[];
    ok(res, rows);
  });

  r.post('/', (req, res) => {
    const term = String(req.body?.term ?? '').trim();
    const category = String(req.body?.category ?? 'hard_skill');
    if (!term) return fail(res, 400, '词条不能为空');
    if (!CATS.includes(category)) return fail(res, 400, `category 必须是 ${CATS.join('/')}`);
    if (db.prepare('SELECT id FROM keywords WHERE term=?').get(term)) return fail(res, 400, '该词条已存在');
    const t = nowIso();
    db.prepare('INSERT INTO keywords (term, category, created_at, updated_at) VALUES (?,?,?,?)').run(term, category, t, t);
    ok(res, { term, category, note: '', totalFrequency: 0 });
  });

  r.put('/:id', (req, res) => {
    const id = Number(req.params.id);
    const cur = db.prepare('SELECT * FROM keywords WHERE id=?').get(id) as { id: number; term: string; category: string; note: string } | undefined;
    if (!cur) return fail(res, 404, '词条不存在');
    const category = String(req.body?.category ?? cur.category);
    if (!CATS.includes(category)) return fail(res, 400, `category 必须是 ${CATS.join('/')}`);
    const note = req.body?.note !== undefined ? String(req.body.note) : cur.note;
    db.prepare('UPDATE keywords SET category=?, note=?, updated_at=? WHERE id=?').run(category, note, nowIso(), id);
    ok(res, { id, term: cur.term, category, note });
  });

  r.delete('/:id', (req, res) => {
    const id = Number(req.params.id);
    db.prepare('DELETE FROM jd_keywords WHERE keyword_id=?').run(id);
    db.prepare('DELETE FROM keyword_hits WHERE keyword_id=?').run(id);
    const info = db.prepare('DELETE FROM keywords WHERE id=?').run(id);
    ok(res, { deleted: info.changes });
  });

  return r;
}
```

```tsx
  const [bullets, setBullets] = useState<string[]>([]);
  const [picked, setPicked] = useState<boolean[]>([]);
  const [busy, setBusy] = useState(false);

  async function run() {
    setBusy(true); setBullets([]);
    try { const r = await api.aiBullets(text); setBullets(r.bullets); setPicked(r.bullets.map(() => true)); }
    catch (e) { onError((e as Error).message); } finally { setBusy(false); }
  }

  async function createItem() {
    const chosen = bullets.filter((_, i) => picked[i]);
    if (!chosen.length) { onError('至少勾选一条'); return; }
    try { await api.createItem({ section: 'work', heading: '', period: '', bullets: chosen }); onDone(); }
    catch (e) { onError((e as Error).message); }
  }

  return (
    <div className="card">
      <b>AI 流水账拆亮点</b>
      <textarea className="wide" rows={5} placeholder="把工作流水账原样贴进来（越具体越好，带数字最佳），AI 帮你拆成 STAR 风格简历亮点" value={text} onChange={e => setText(e.target.value)} />
      <div className="row"><button className="primary" onClick={run} disabled={busy || text.trim().length < 10}>{busy ? 'AI 拆解中…' : '✨ AI 拆亮点'}</button></div>
      {bullets.length > 0 && (
        <div>
          {bullets.map((b, i) => (
            <label key={i}><input type="checkbox" checked={picked[i]} onChange={() => setPicked(picked.map((x, j) => (j === i ? !x : x)))} /> {b}</label>
          ))}
          <div className="row"><button className="primary" onClick={createItem}>写入为新条目</button></div>
        </div>
      )}
    </div>
  );
}
```

```tsx
function JdsPanel() {

  const [company, setCompany] = useState('');

  const [raw, setRaw] = useState('');

function KeywordsPanel() {
```
