# JobCoach 计划①：M1 底座 + M2 个人履历 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 搭建全栈 TypeScript 跑通骨架（一条命令启动），并实现界面①个人履历：结构化条目的增删改 + 每次保存自动留版本 + 一键恢复旧版。

**Architecture:** 单包双端：`server/`（Express + Node 24 内置 `node:sqlite` 单文件库）、`web/`（React+Vite），开发时 Vite 代理 `/api` 到 3001；所有 M2~M9 数据表在骨架期一次建齐（建表成本≈0，后续里程碑只加路由）。核心机制：每次写履历条目后调用 `takeVersion` 存全量快照，超过 50 版自动裁剪最旧。

**Tech Stack:** Node.js 24（用内置 `node:sqlite`，无原生三方依赖）、TypeScript、Express 4、React 18、Vite 5、vitest 2、supertest 7。设计依据：`docs/superpowers/specs/2026-10-26-jobcoach-design-v2.md`。（2026-10-26 执行时修订：因本机 Node 为 v24，better-sqlite3 全部替换为 node:sqlite。）

```ts
import { DatabaseSync } from 'node:sqlite';
import fs from 'node:fs';
import path from 'node:path';

export const SECTIONS = ['work', 'project', 'education', 'skill', 'award'] as const;
export type Section = (typeof SECTIONS)[number];

export function nowIso(): string { return new Date().toISOString(); }

export function openDb(file = path.resolve('data/jobcoach.db')): DatabaseSync {
  fs.mkdirSync(path.dirname(file), { recursive: true });
  const db = new DatabaseSync(file);
  db.exec('PRAGMA journal_mode = WAL;');
  migrate(db);
  return db;
}

export interface ItemRow {
  id: number; section: string; heading: string; period: string;
  bullets: string; sort_order: number; created_at: string; updated_at: string;
}

export function migrate(db: DatabaseSync): void {
  db.exec(`
  CREATE TABLE IF NOT EXISTS profile_items (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    section TEXT NOT NULL,
    heading TEXT NOT NULL DEFAULT '',
    period TEXT NOT NULL DEFAULT '',
    bullets TEXT NOT NULL DEFAULT '[]',
    sort_order INTEGER NOT NULL DEFAULT 0,
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
  );
  CREATE TABLE IF NOT EXISTS profile_versions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    item_id INTEGER NOT NULL,
    snapshot TEXT NOT NULL,
    trigger TEXT NOT NULL DEFAULT 'manual',
    created_at TEXT NOT NULL
  );
  CREATE TABLE IF NOT EXISTS materials (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL DEFAULT '',
    content TEXT NOT NULL DEFAULT '',
    tags TEXT NOT NULL DEFAULT '[]',
    source TEXT NOT NULL DEFAULT 'manual',
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
  );
  CREATE TABLE IF NOT EXISTS jds (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    company TEXT NOT NULL DEFAULT '',
    title TEXT NOT NULL DEFAULT '',
    raw_text TEXT NOT NULL DEFAULT '',
    summary TEXT NOT NULL DEFAULT '{}',
    analyzed INTEGER NOT NULL DEFAULT 0,
    recommended TEXT NOT NULL DEFAULT '[]',
    chosen_items TEXT NOT NULL DEFAULT '[]',
    overrides TEXT NOT NULL DEFAULT '{}',
    status TEXT NOT NULL DEFAULT 'draft',
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
  );
  CREATE TABLE IF NOT EXISTS keywords (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    term TEXT NOT NULL UNIQUE,
    category TEXT NOT NULL DEFAULT 'hard_skill',
    note TEXT NOT NULL DEFAULT '',
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
  );
  CREATE TABLE IF NOT EXISTS jd_keywords (
    jd_id INTEGER NOT NULL,
    keyword_id INTEGER NOT NULL,
    frequency INTEGER NOT NULL DEFAULT 1,
    PRIMARY KEY (jd_id, keyword_id)
  );
  CREATE TABLE IF NOT EXISTS keyword_hits (
    jd_id INTEGER NOT NULL,
    keyword_id INTEGER NOT NULL,
    status TEXT NOT NULL DEFAULT 'missing',
    PRIMARY KEY (jd_id, keyword_id)
  );
  CREATE TABLE IF NOT EXISTS coach_runs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    jd_id INTEGER NOT NULL,
    item_id INTEGER NOT NULL,
    original_bullets TEXT NOT NULL DEFAULT '[]',
    suggestions TEXT NOT NULL DEFAULT '[]',
    backfilled INTEGER NOT NULL DEFAULT 0,
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
  );
  CREATE TABLE IF NOT EXISTS plans (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    jd_id INTEGER NOT NULL,
    weeks TEXT NOT NULL DEFAULT '[]',
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
  );
  CREATE TABLE IF NOT EXISTS questions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    jd_id INTEGER NOT NULL,
    category TEXT NOT NULL,
    question TEXT NOT NULL,
    hint TEXT NOT NULL DEFAULT '',
    based_on TEXT NOT NULL DEFAULT '',
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
  );
  CREATE TABLE IF NOT EXISTS mock_chats (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    question_id INTEGER NOT NULL,
    messages TEXT NOT NULL DEFAULT '[]',
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL
  );
  CREATE TABLE IF NOT EXISTS settings (
    key TEXT PRIMARY KEY,
    value TEXT NOT NULL
  );`);
}

export function takeVersion(db: DatabaseSync, itemId: number, trigger: string): number {
  const row = db.prepare('SELECT heading, bullets FROM profile_items WHERE id = ?').get(itemId) as Pick<ItemRow, 'heading' | 'bullets'> | undefined;
  if (!row) throw new Error('takeVersion: 条目不存在 id=' + itemId);
  const snapshot = JSON.stringify({ heading: row.heading, bullets: JSON.parse(row.bullets) });
  const info = db.prepare('INSERT INTO profile_versions (item_id, snapshot, trigger, created_at) VALUES (?, ?, ?, ?)')
    .run(itemId, snapshot, trigger, nowIso());
  trimVersions(db, itemId);
  return Number(info.lastInsertRowid);
}

export function trimVersions(db: DatabaseSync, itemId: number, keep = 50): number {
  const stale = db.prepare('SELECT id FROM profile_versions WHERE item_id = ? ORDER BY id DESC LIMIT -1 OFFSET ?').all(itemId, keep) as { id: number }[];
  const del = db.prepare('DELETE FROM profile_versions WHERE id = ?');
  for (const v of stale) del.run(v.id);
  return stale.length;
}
```

```ts
import { Router } from 'express';
import { DatabaseSync } from 'node:sqlite';
import { ok, fail } from '../http';
import { SECTIONS, nowIso, takeVersion, ItemRow } from '../db';

interface ItemDto { id: number; section: string; heading: string; period: string; bullets: string[]; sort_order: number; created_at: string; updated_at: string }

function parseItem(row: ItemRow): ItemDto {
  return { ...row, bullets: JSON.parse(row.bullets) };
}

function validateBullets(body: any): boolean {
  const b = body?.bullets ?? [];
  return Array.isArray(b) && b.every((x: unknown) => typeof x === 'string');
}

export function profileRouter(db: DatabaseSync): Router {
  const r = Router();

  r.get('/items', (_req, res) => {
    const rows = db.prepare('SELECT * FROM profile_items ORDER BY section, sort_order, id').all() as ItemRow[];
    ok(res, rows.map(parseItem));
  });

  r.post('/items', (req, res) => {
    const body = req.body ?? {};
    if (!SECTIONS.includes(body.section)) return fail(res, 400, `section 必须是 ${SECTIONS.join('/')} 之一`);
    if (!validateBullets(body)) return fail(res, 400, 'bullets 必须是字符串数组');
    const t = nowIso();
    const info = db.prepare('INSERT INTO profile_items (section, heading, period, bullets, sort_order, created_at, updated_at) VALUES (?,?,?,?,?,?,?)')
      .run(body.section, body.heading ?? '', body.period ?? '', JSON.stringify(body.bullets ?? []), body.sort_order ?? 0, t, t);
    const id = Number(info.lastInsertRowid);
    takeVersion(db, id, 'create');
    ok(res, parseItem(db.prepare('SELECT * FROM profile_items WHERE id=?').get(id) as ItemRow));
  });

  r.put('/items/:id', (req, res) => {
    const id = Number(req.params.id);
    if (!db.prepare('SELECT id FROM profile_items WHERE id=?').get(id)) return fail(res, 404, '条目不存在');
    const body = req.body ?? {};
    if (body.bullets !== undefined && !validateBullets(body)) return fail(res, 400, 'bullets 必须是字符串数组');
    db.prepare('UPDATE profile_items SET heading=?, period=?, bullets=?, updated_at=? WHERE id=?')
      .run(body.heading ?? '', body.period ?? '', JSON.stringify(body.bullets ?? []), nowIso(), id);
    takeVersion(db, id, 'manual');
    ok(res, parseItem(db.prepare('SELECT * FROM profile_items WHERE id=?').get(id) as ItemRow));
  });

  r.delete('/items/:id', (req, res) => {
    const id = Number(req.params.id);
    db.prepare('DELETE FROM profile_versions WHERE item_id=?').run(id);
    const info = db.prepare('DELETE FROM profile_items WHERE id=?').run(id);
    ok(res, { deleted: info.changes });
  });

  r.get('/items/:id/versions', (req, res) => {
    const id = Number(req.params.id);
    const rows = db.prepare('SELECT * FROM profile_versions WHERE item_id=? ORDER BY id DESC').all(id) as { id: number; item_id: number; snapshot: string; trigger: string; created_at: string }[];
    ok(res, rows.map(v => ({ ...v, snapshot: JSON.parse(v.snapshot) })));
  });

  r.post('/items/:id/restore', (req, res) => {
    const id = Number(req.params.id);
    const vid = Number(req.body?.version_id);
    const ver = db.prepare('SELECT snapshot FROM profile_versions WHERE id=? AND item_id=?').get(vid, id) as { snapshot: string } | undefined;
    if (!ver) return fail(res, 404, '条目或版本不存在');
    const snap = JSON.parse(ver.snapshot) as { heading: string; bullets: string[] };
    db.prepare('UPDATE profile_items SET heading=?, bullets=?, updated_at=? WHERE id=?')
      .run(snap.heading, JSON.stringify(snap.bullets ?? []), nowIso(), id);
    takeVersion(db, id, 'restore');
    ok(res, { restored_version: vid });
  });

  return r;
}
```

import { useState } from 'react';
import { ProfileItem } from '../types';
import { api } from '../api';
import VersionsDrawer from './VersionsDrawer';

export default function ItemCard({ item, onChanged, onError }: {
  item: ProfileItem; onChanged: () => void; onError: (m: string) => void;
}) {
  const [heading, setHeading] = useState(item.heading);
  const [period, setPeriod] = useState(item.period);
  const [bullets, setBullets] = useState<string[]>(item.bullets.length ? item.bullets : ['']);
  const [saving, setSaving] = useState(false);
  const [showVersions, setShowVersions] = useState(false);

  async function save() {
    setSaving(true);
    try {
      await api.updateItem(item.id, { heading, period, bullets: bullets.map(b => b.trim()).filter(b => b.length) });
      onChanged();
    } catch (e) { onError((e as Error).message); } finally { setSaving(false); }
  }

  async function remove() {
    if (!confirm(`确认删除「${heading || '此条目'}」？其历史版本也会被删除。`)) return;
    try { await api.deleteItem(item.id); onChanged(); } catch (e) { onError((e as Error).message); }
  }

  return (
    <div className="card">
      <div className="row">
        <input className="wide" placeholder="抬头：公司 - 岗位" value={heading} onChange={e => setHeading(e.target.value)} />
        <input style={{ width: 160 }} placeholder="起止时间（可选）" value={period} onChange={e => setPeriod(e.target.value)} />
      </div>
      {bullets.map((b, i) => (
        <div key={i} className="row">
          <textarea className="wide" rows={2} placeholder={`亮点 ${i + 1}：做了什么 → 怎么做 → 结果（数字）`} value={b}
            onChange={e => setBullets(bullets.map((x, j) => (j === i ? e.target.value : x)))} />
          <button title="删除这条亮点" onClick={() => setBullets(bullets.filter((_, j) => j !== i))}>✕</button>
        </div>
      ))}
      <div className="row">
        <button onClick={() => setBullets([...bullets, ''])}>+ 加一条亮点</button>
        <span className="wide" />
        <button onClick={() => setShowVersions(v => !v)}>历史版本</button>
        <button className="primary" onClick={save} disabled={saving}>{saving ? '保存中…' : '保存'}</button>
        <button className="danger" onClick={remove}>删除条目</button>
      </div>
      {showVersions && (
        <VersionsDrawer item={item} onError={onError}
          onRestored={() => { setShowVersions(false); onChanged(); }} />
      )}
    </div>
  );
}
